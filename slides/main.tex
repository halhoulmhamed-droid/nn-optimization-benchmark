\renewcommand{\sectionlabel}{01 / Question}
\begin{frame}[t]{Deux questions distinctes}
\framesubtitle{Apprendre une fonction, puis choisir une durée physique}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\card{Apprentissage}{%
Paramètres du réseau : $\theta\in\mathbb R^{49}$.
\[\min_\theta\mathcal L(\theta).\]
La perte est généralement non convexe.}
\end{column}
\begin{column}{0.48\textwidth}
\card{Décision}{%
Durée physique : $p\in[0.02,0.20]$.
\[\min_p J_\lambda(p).\]
L'objectif physique est fortement convexe.}
\end{column}
\end{columns}
\vspace{0.35cm}
\textbf{Association.} L'erreur de sensibilité classe-t-elle mieux la mauvaise décision ?\par\vspace{0.18cm}
\textbf{Comparaison.} La supervision valeurs et dérivées améliore-t-elle le score moyen, et sous quels coûts ?
\vfill
\takeaway{À distinguer :}{convexité en $p$, apprentissage en $\theta$ et comparaison des méthodes.}
\end{frame}

\renewcommand{\sectionlabel}{02 / Modèle physique}
\begin{frame}[t]{Un oracle thermique analytique}
\framesubtitle{Deux modes de l'équation de la chaleur ; variables sans dimension}
\[
T(p)=\frac{e^{-\pi^2p}}{\sqrt2}+0.2e^{-4\pi^2p},
\qquad J_\lambda(p)=T(p)+\lambda p.
\]
Capteur $x=1/4$ de $u_t=u_{xx}$, avec des bords homogènes.
\[J'_\lambda=T'+\lambda,\qquad 0<m=T''(b)\leq J''_\lambda.\]
\card{Référence physique M0}{%
$\lambda_b=-T'(b)<\lambda_a=-T'(a)$.\par\vspace{3pt}
$p_{\rm ref}=b$ si $\lambda\leq\lambda_b$ ; $p_{\rm ref}=a$ si $\lambda\geq\lambda_a$.\par\vspace{3pt}
Sinon : dichotomie sur l'unique racine intérieure.}
\vfill
\takeaway{Limite numérique :}{M0 est une référence flottante ; ses bornes restent conditionnelles.}
\end{frame}

\renewcommand{\sectionlabel}{03 / Réseau et supervision}
\begin{frame}[t]{Un réseau scalaire à 49 paramètres}
\framesubtitle{Architecture tanh 1--16--1 ; dérivée par rapport à la durée physique}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\card{Normalisation et paramètres}{%
\[z=\frac{2(p-a)}{b-a}-1,\quad\alpha_N=\frac{100}{9}.\]
\[\theta=(w,b,v,d).\]
\[16+16+16+1=49.\]
$c_{\rm out}=0$, $s_{\rm out}=1$.}
\end{column}
\begin{column}{0.49\textwidth}
\card{Valeur et sensibilité}{%
\[h_i=\tanh(w_i z+b_i).\]
\[\widehat T=d+\sum_i v_i h_i.\]
\[\widehat T'(p)=\alpha_N\sum_i v_iw_i(1-h_i^2).\]}
\end{column}
\end{columns}
\vfill
\takeaway{Point de vigilance :}{l'architecture n'impose ni monotonie ni convexité.}
\end{frame}

\begin{frame}[t]{Apprendre les valeurs et les dérivées}
\framesubtitle{Même architecture ; deux pertes avec des échelles fixées par l'oracle}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\card{M1 : valeurs seules}{%
\[\mathcal L_1=\frac1n\sum_j\frac{(\widehat T-T)^2}{S_0^2}.\]
\[S_0=T(a)-T(b)\simeq0.57295.\]}
\end{column}
\begin{column}{0.49\textwidth}
\card{M2 : valeurs et sensibilités}{%
\[\mathcal L_2=\mathcal L_1+\frac1n\sum_j\frac{(\widehat T'-T')^2}{S_1^2}.\]
\[S_1=-T'(a)\simeq9.31369.\]}
\end{column}
\end{columns}
\vspace{0.32cm}
Coefficient différentiel 1 ; aucun facteur $1/2$ ; aucune régularisation.\par\vspace{0.14cm}
Gradients analytiques des 49 paramètres.
\vfill
\takeaway{Échelles fixes :}{aucune estimation de $S_0$ ou $S_1$ sur les sites diagnostiques.}
\end{frame}

\begin{frame}[t]{Des garanties sous hypothèses}
\framesubtitle{Bornes classiques ; distinction entre théorie et contrôle flottant}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\card{Erreur uniforme}{%
\[R\leq2\varepsilon+\alpha_{\rm opt}.\]
Erreur uniforme et défaut global d'optimisation majorés.\par\vspace{4pt}
Une RMSE ne fournit pas $\varepsilon$.}
\end{column}
\begin{column}{0.48\textwidth}
\card{Optimalité physique}{%
\[|p-p^\star|\leq r/m.\]
\[R\leq r^2/(2m).\]
$r$ : résidu KKT physique, avec les signes adaptés aux frontières.}
\end{column}
\end{columns}
\vspace{0.28cm}
Décision neuronale : meilleur des 1025 candidats, sans raffinement.\par\vspace{0.12cm}
Aucune convexité du réseau supposée.
\vfill
\takeaway{Pas de certificat numérique :}{les erreurs flottantes totales ne sont pas majorées.}
\end{frame}
